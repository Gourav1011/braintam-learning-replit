2. DATABASE TABLES
==================

## leads

lead_id TEXT PRIMARY KEY
student_name TEXT
parent_phone TEXT
grade INT
assigned_mentor_id TEXT DEFAULT NULL
status TEXT DEFAULT 'Fresh'
deployment_cycle_id TEXT DEFAULT NULL
created_at TIMESTAMP
updated_at TIMESTAMP

Status:
Fresh
Ignite_Active
Mastery_Converted
Lost

## batches

batch_id TEXT PRIMARY KEY
course_id TEXT
batch_name TEXT
start_date DATE
end_date DATE
status TEXT DEFAULT 'Upcoming'
max_students INT DEFAULT NULL
created_at TIMESTAMP

Status:
Upcoming
Active
Completed

## payment_links

payment_link_id TEXT PRIMARY KEY
program_type TEXT
grade INT
amount DECIMAL
linked_course_id TEXT
is_active BOOLEAN DEFAULT TRUE
created_at TIMESTAMP

## enrollments

enrollment_id TEXT PRIMARY KEY
lead_id TEXT
course_id TEXT
batch_id TEXT DEFAULT NULL
enrollment_type TEXT
created_at TIMESTAMP

## mentor_conversion_logs

id TEXT PRIMARY KEY
lead_id TEXT
mentor_id TEXT
payment_id TEXT
payment_link_id TEXT
course_id TEXT
amount DECIMAL
created_at TIMESTAMP
