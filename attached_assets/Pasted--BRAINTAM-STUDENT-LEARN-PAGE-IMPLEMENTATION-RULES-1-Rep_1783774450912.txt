/* =========================================================

========================================================= */

function clampProgress(
  progress?: number | null
) {
  return Math.min(
    100,
    Math.max(
      0,
      Number(progress || 0)
    )
  );
}

function getFirstName(
  student?: Student | null
) {
  return (
    student?.firstName ||
    student?.name
      ?.trim()
      .split(/\s+/)[0] ||
    "Student"
  );
}

function getSubjectVisual(
  subjectName: string
) {
  const normalized =
    subjectName
      .toLowerCase()
      .trim();

  if (
    normalized.includes("math")
  ) {
    return {
      emoji: "123",
      className: "subject-blue"
    };
  }

  if (
    normalized.includes("science") ||
    normalized.includes("physics") ||
    normalized.includes("chemistry") ||
    normalized.includes("biology")
  ) {
    return {
      emoji: "⚗",
      className: "subject-green"
    };
  }

  if (
    normalized.includes("english")
  ) {
    return {
      emoji: "📖",
      className: "subject-orange"
    };
  }

  if (
    normalized.includes("hindi")
  ) {
    return {
      emoji: "अ",
      className: "subject-pink"
    };
  }

  if (
    normalized.includes("computer")
  ) {
    return {
      emoji: "🖥",
      className: "subject-purple"
    };
  }

  if (
    normalized.includes("social")
  ) {
    return {
      emoji: "🌍",
      className: "subject-cyan"
    };
  }

  if (
    normalized.includes("sanskrit")
  ) {
    return {
      emoji: "ॐ",
      className: "subject-yellow"
    };
  }

  if (
    normalized.includes("french")
  ) {
    return {
      emoji: "🇫🇷",
      className: "subject-blue"
    };
  }

  return {
    emoji: "📚",
    className: "subject-purple"
  };
}

function formatCompletionDate(
  value?: string | null
) {
  if (!value) {
    return "Course completed";
  }

  const date = new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return "Course completed";
  }

  return new Intl.DateTimeFormat(
    "en-IN",
    {
      day: "numeric",
      month: "short",
      year: "numeric"
    }
  ).format(date);
}

export default function StudentLearnPage() {
  const {
    data,
    isLoading,
    error
  } = useExistingStudentLearnData();

  const [
    activeTab,
    setActiveTab
  ] =
    useState<MainLearnTab>(
      "current"
    );

  const [
    search,
    setSearch
  ] = useState("");

  const [
    selectedSubject,
    setSelectedSubject
  ] = useState("all");

  const filteredSubjects =
    useMemo(() => {
      if (!data) {
        return [];
      }

      const query =
        search
          .trim()
          .toLowerCase();

      return data.currentSubjects
        .filter((subject) => {
          const matchesSearch =
            !query ||
            subject.name
              .toLowerCase()
              .includes(query);

          const matchesFilter =
            selectedSubject ===
              "all" ||
            subject.id ===
              selectedSubject;

          return (
            matchesSearch &&
            matchesFilter
          );
        });
    }, [
      data,
      search,
      selectedSubject
    ]);

  const filteredCompletedCourses =
    useMemo(() => {
      if (!data) {
        return [];
      }

      const query =
        search
          .trim()
          .toLowerCase();

      return data.completedCourses
        .filter((course) => {
          return (
            !query ||
            course.title
              .toLowerCase()
              .includes(query) ||
            course.grade
              .toLowerCase()
              .includes(query) ||
            course.academicYear
              ?.toLowerCase()
              .includes(query)
          );
        });
    }, [
      data,
      search
    ]);

  if (isLoading) {
    return (
      <LearnPageSkeleton />
    );
  }

  if (error) {
    return (
      <LearnErrorState />
    );
  }

  if (!data) {
    return (
      <LearnEmptyState />
    );
  }

  return (
    <div className="learn-page">
      <LearnTopBar
        student={data.student}
      />

      <main className="learn-content">
        <header className="learn-header">
          <div>
            <h1>
              My Learning
              <span
                aria-hidden="true"
              >
                📖
              </span>
            </h1>

            <p>
              Explore your subjects
              and keep learning
            </p>
          </div>

          <div
            className="learn-header-art"
            aria-hidden="true"
          >
            <span className="header-planet">
              🪐
            </span>

            <span className="header-rocket">
              🚀
            </span>
          </div>
        </header>

        <LearnMainTabs
          activeTab={activeTab}
          completedCount={
            data
              .completedCourses
              .length
          }
          onChange={setActiveTab}
        />

        <SearchAndFilter
          search={search}
          onSearchChange={
            setSearch
          }
          activeTab={activeTab}
          subjects={
            data.currentSubjects
          }
          selectedSubject={
            selectedSubject
          }
          onSubjectChange={
            setSelectedSubject
          }
        />

        {activeTab ===
        "current" ? (
          <CurrentLearning
            data={data}
            subjects={
              filteredSubjects
            }
          />
        ) : (
          <CompletedLearning
            courses={
              filteredCompletedCourses
            }
          />
        )}
      </main>

      <AskMentorButton />

      <StudentBottomNav
        active="learn"
      />
    </div>
  );
}

    (
      value: string

    string;

    ) => void;
}) {
  return (
    <section
      className=
        "learn-search-row"
    >
      <label
        className=
          "learn-search"
      >

          size={20}
        />

        <input
          value={search}
          onChange={(event) =>
            onSearchChange(
              event.target.value
            )
          }
          placeholder={
            activeTab ===
            "current"
              ? "Search subjects or lessons..."
              : "Search completed courses..."
          }

      </label>

      {activeTab ===
        "current" && (
        <label
          className=
            "subject-filter"

                event

function IgniteProgramCard({
  program
}: {
  program: IgniteProgram;
}) {
  const progress =
    clampProgress(
      (
        program.completedDays /
        Math.max(
          program.totalDays,
          1
        )
      ) *
        100
    );

  return (
    <section
      className=
        "ignite-program"
    >
      <div
        className=
          "ignite-main"
      >
        <p
          className=
            "section-label green"
        >
          Current Program
        </p>

        <div
          className=
            "ignite-title-row"
        >
          <span
            className=
              "ignite-icon"
            aria-hidden="true"
          >
            🚀
          </span>

          <div>
            <div
              className=
                "title-with-badge"
            >
              <h2>
                {program.name}
              </h2>

              <span
                className=
                  "grade-badge"
              >
                {program.grade}
              </span>
            </div>

            <p>
              Day{" "}
              {
                program.completedDays
              }{" "}
              of{" "}
              {
                program.totalDays
              }{" "}
              completed
            </p>
          </div>
        </div>

        <div
          className=
            "ignite-days"
          aria-label=
            "Program days"
        >
          {program.days.map(
            (day) => (
              <div
                key={
                  day.day
                }
                className={[
                  "ignite-day",
                  day.completed
                    ? "complete"
                    : "",
                  day.locked
                    ? "locked"
                    : ""
                ]
                  .filter(
                    Boolean
                  )
                  .join(" ")}
              >
                {day.completed ? (
                  <Check
                    size={15}
                  />
                ) : day.locked ? (
                  <LockKeyhole
                    size={13}
                  />
                ) : (
                  <span
                    className=
                      "empty-day"
                  />
                )}

                D{day.day}
              </div>
            )
          )}
        </div>

        <div
          className=
            "ignite-progress-row"
        >
          <div
            className=
              "progress-track"
          >
            <div
              className=
                "progress-fill green"
              style={{
                width:
                  `${progress}%`
              }}
            />
          </div>

          <strong>
            {Math.round(
              progress
            )}
            %
          </strong>
        </div>
      </div>

      <div
        className=
          "ignite-rocket-art"
        aria-hidden="true"
      >
        🚀
      </div>

      <div
        className=
          "next-session"
      >
        <div
          className=
            "next-session-title"
        >
          <CalendarDays
            size={19}
          />

          <strong>
            Next session
          </strong>
        </div>

        {program.nextSession ? (
          <p>
            {program
              .nextSession
              .label ||
              [
                program
                  .nextSession
                  .date,
                program
                  .nextSession
                  .time
              ]
                .filter(
                  Boolean
                )
                .join(
                  " • "
                )}
          </p>
        ) : (
          <p>
            No session
            scheduled yet
          </p>
        )}

        <a
          href={
            `/courses/${program.id}`
          }
          className=
            "primary-button green"
        >
          View Program

          <ChevronRight
            size={18}
          />
        </a>
      </div>
    </section>
  );
}

function SubjectCard({
  subject
}: {
  subject: Subject;
}) {
  const visual =
    getSubjectVisual(
      subject.name
    );

  const progress =
    clampProgress(
      subject.progress
    );

  const hasStarted =
    progress > 0;

  const actionHref =
    subject.lastLessonId
      ? `/lessons/${subject.lastLessonId}`
      : `/subjects/${subject.id}`;

  return (
    <article
      className={
        `subject-card ${visual.className}`
      }
    >
      <div
        className=
          "subject-icon"
        aria-hidden="true"
      >
        {visual.emoji}
      </div>

      <h3>
        {subject.name}
      </h3>

      <p>
        {subject.lessonCount}{" "}
        {subject.lessonCount ===
        1
          ? "Lesson"
          : "Lessons"}
      </p>

      <div
        className=
          "subject-progress"
      >
        <div
          className=
            "progress-track"
        >
          <div
            className=
              "progress-fill"
            style={{
              width:
                `${progress}%`
            }}
          />
        </div>
      </div>

      <span
        className=
          "subject-status"
      >
        {hasStarted
          ? `${Math.round(
              progress
            )}% Complete`
          : "Not Started"}
      </span>

      <a
        href={actionHref}
        className=
          "subject-button"
      >
        {hasStarted
          ? "Continue"
          : "Start Learning"}

        <ChevronRight
          size={15}
        />
      </a>
    </article>
  );
}

function StudentBottomNav({
  active
}: {
  active:
    | "home"
    | "classes"
    | "learn"
    | "tasks"
    | "rewards";
}) {
  const items = [
    {
      id: "home",
      label: "Home",
      href: "/dashboard",
      icon: Home
    },
    {
      id: "classes",
      label: "Classes",
      href:
        "/live-classes",
      icon: Video
    },
    {
      id: "learn",
      label: "Learn",
      href: "/courses",
      icon: BookOpen
    },
    {
      id: "tasks",
      label: "Tasks",
      href: "/tasks",
      icon:
        ClipboardTaskIcon
    },
    {
      id: "rewards",
      label: "Rewards",
      href: "/rewards",
      icon: Trophy
    }
  ] as const;

  return (
    <nav
      className=
        "student-bottom-nav"
      aria-label=
        "Student navigation"
    >
      {items.map(
        (item) => {
          const Icon =
            item.icon;

          const isActive =
            active ===
            item.id;

          return (
            <a
              key={
                item.id
              }
              href={
                item.href
              }
              className={
                isActive
                  ? "bottom-nav-item active"
                  : "bottom-nav-item"
              }
              aria-current={
                isActive
                  ? "page"
                  : undefined
              }
            >
              <span
                className=
                  "bottom-nav-icon"
              >
                <Icon
                  size={22}
                />
              </span>

              <span>
                {
                  item.label
                }
              </span>
            </a>
          );
        }
      )}
    </nav>
  );
}
