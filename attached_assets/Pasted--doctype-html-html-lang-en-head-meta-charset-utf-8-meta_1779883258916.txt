<script>
/* Grade-specific demo (6-day mix of tricks + core) + full subjects
   - Demo shows grade-specific mixed chips (e.g. "Math — Tricks", "Science — Activity")
   - Full shows real syllabus subjects
   - Demo CTAs stay side-by-side on mobile
   - Page bottom whitespace reduced by smaller paddings
*/

(function(){
  const gradeBtns = Array.from(document.querySelectorAll('.grade-btn'));

  const demoTitle = document.getElementById('demoTitle');
  const demoLead  = document.getElementById('demoLead');
  const demoChips = document.getElementById('demoChips');
  const demoPrice = document.getElementById('demoPrice');
  const demoFreq  = document.getElementById('demoFreq');
  const demoCta   = document.getElementById('demoCta');

  const fullTitle = document.getElementById('fullTitle');
  const fullLead  = document.getElementById('fullLead');
  const fullChips = document.getElementById('fullChips');
  const fullPrice = document.getElementById('fullPrice');
  const fullFreq  = document.getElementById('fullFreq');
  const fullCta   = document.getElementById('fullCta');

  // grade-specific subject mapping (demo = 6-day mix: core + tricks; full = syllabus)
  const data = {
    1:{ demoSubjects:['Counting & Numbers (Tricks)','Phonics & Reading','Shapes & Patterns'], fullSubjects:['Numbers','English','EVS'], fullPrice:'26,999', demoLead:'6-Day camp: basics + fun tricks to start strong.' },
    2:{ demoSubjects:['Addition & Tricks','Reading Skills','Reasoning Games'], fullSubjects:['Math','English','EVS'], fullPrice:'27,999', demoLead:'6-Day camp: build speed & confidence.' },
    3:{ demoSubjects:['Tables & Tricks','Science Basics','Comprehension Tips'], fullSubjects:['Math','Science','English'], fullPrice:'28,999', demoLead:'6-Day camp: strengthen fundamentals.' },
    4:{ demoSubjects:['Fractions Tricks','Science Experiments','Grammar Boost'], fullSubjects:['Math','Science','English'], fullPrice:'29,999', demoLead:'6-Day camp: problem-solving focus.' },
    5:{ demoSubjects:['High-weight Math Tricks','Hands-on Science','Vocabulary & Writing'], fullSubjects:['Math','Science','Reasoning'], fullPrice:'30,999', demoLead:'6-Day camp: bridge to middle school.' },
    6:{ demoSubjects:['Algebra Basics (Tricks)','Physics Intros','Comprehension'], fullSubjects:['Math','Science','English'], fullPrice:'31,999', demoLead:'6-Day camp: foundation for higher classes.' },
    7:{ demoSubjects:['Algebra Tricks','Mechanics Intro','Critical Reading'], fullSubjects:['Math','Science','English'], fullPrice:'32,999', demoLead:'6-Day camp: stronger problem solving.' },
    8:{ demoSubjects:['Exam-focused Math Tricks','Physics Basics','Chemistry Basics'], fullSubjects:['Math','Science(PCB)','English'], fullPrice:'33,999', demoLead:'6-Day camp: boards & olympiad readiness.' },
    9:{ demoSubjects:['Grade 9 High-yield Math','Physics Tricks','Chemistry Tricks'], fullSubjects:['Math','Science(PCB)','English'], fullPrice:'34,999', demoLead:'6-Day camp: revision & tests.' },
    10:{ demoSubjects:['Board Revision Tricks','Quick Physics','Chemistry Shortcuts'], fullSubjects:['Math','Science','English'], fullPrice:'35,999', demoLead:'6-Day camp: board-focused high-yield topics.' },
    11:{ demoSubjects:['Calculus Beginnings','Mechanics Tricks','Organic Intros'], fullSubjects:['PCM/PCB/Commerce','English'], fullPrice:'36,999', demoLead:'6-Day camp: bridge to class 11 concepts.' },
    12:{ demoSubjects:['Exam Revision Tricks','PYQ Practice','Strategy & Shortcuts'], fullSubjects:['PCM/PCB/Commerce','English'], fullPrice:'37,999', demoLead:'6-Day camp: final push for boards & entrances.' }
  };

  function setChips(container, arr){
    container.innerHTML = '';
    arr.forEach(it=>{
      const el = document.createElement('div');
      el.className = 'chip';
      el.textContent = it;
      container.appendChild(el);
    });
  }

  function fadeUpdate(elems, updater){
    elems.forEach(el => { el.classList.remove('in'); el.classList.add('fade'); });
    setTimeout(()=>{
      updater();
      elems.forEach(el => { void el.offsetWidth; el.classList.remove('fade'); el.classList.add('in'); });
    }, 120);
  }

  function updateForGrade(g){
    const info = data[g] || data[1];

    fadeUpdate([demoTitle, demoLead, demoFreq, demoPrice, demoChips], ()=>{
      demoTitle.textContent = '6-Day Core Concepts — Demo';
      demoLead.textContent = info.demoLead || '6-Day tricks + core practice.';
      setChips(demoChips, info.demoSubjects || ['Math Tricks','Science Tricks','English Tricks']);
      demoPrice.textContent = '₹99';
      demoFreq.textContent = '5–6 classes / week';
      demoCta.setAttribute('href', '/enroll?grade=' + encodeURIComponent(g) + '&variant=demo');
    });

    fadeUpdate([fullTitle, fullLead, fullFreq, fullPrice, fullChips], ()=>{
      fullTitle.textContent = info.fullSubjects ? ('Live Full Syllabus — Grade ' + g) : ('Live Full Syllabus — Grade ' + g);
      fullLead.textContent = 'Complete syllabus with tests & mentor support.';
      setChips(fullChips, info.fullSubjects || ['Math','English']);
      fullPrice.textContent = '₹' + (info.fullPrice || '');
      fullFreq.textContent = '5–6 classes / week';
      fullCta.setAttribute('href', '/enroll?grade=' + encodeURIComponent(g) + '&variant=full');
    });
  }

  // init
  updateForGrade('1');

  gradeBtns.forEach((btn, idx)=>{
    btn.addEventListener('click', function(){
      gradeBtns.forEach(b=>b.setAttribute('aria-selected','false'));
      this.setAttribute('aria-selected','true');
      updateForGrade(this.dataset.grade);
      try{ this.scrollIntoView({behavior:'smooth', block:'nearest', inline:'nearest'}); }catch(e){}
    });
    btn.addEventListener('keydown', function(e){
      if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); this.click(); }
      if(e.key === 'ArrowRight' || e.key === 'ArrowDown'){ e.preventDefault(); const next = gradeBtns[(idx+1)%gradeBtns.length]; next.focus(); }
      if(e.key === 'ArrowLeft' || e.key === 'ArrowUp'){ e.preventDefault(); const prev = gradeBtns[(idx-1+gradeBtns.length)%gradeBtns.length]; prev.focus(); }
    });
  });

  // wys: small "Why this demo?" opens minimal inline modal (non-blocking)
  document.getElementById('whyDemo').addEventListener('click', function(e){
    e.preventDefault();
    alert('6-Day demo includes:\n• Quick tricks to improve speed\n• 6 short sessions with practice\n• A sample of full-course teaching style');
  });

  // enroll tracking best-effort
  document.addEventListener('click', function(e){
    const a = e.target.closest('a.btn, a.btn.primary, a.primary, a');
    if(!a) return;
    const href = a.getAttribute('href') || '';
    if(!href.includes('enroll')) return;
    const params = new URLSearchParams(href.split('?')[1] || '');
    const grade = params.get('grade') || '';
    const variant = params.get('variant') || '';
    const payload = {event:'enroll_click',grade,variant,ts:Date.now()};
    try{
      if(navigator.sendBeacon){
        const b = new Blob([JSON.stringify(payload)], {type:'application/json'});
        navigator.sendBeacon('/api/track', b);
      } else {
        fetch('/api/track', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(payload),keepalive:true}).catch(()=>{});
      }
    }catch(e){}
  });

})();
</script>
